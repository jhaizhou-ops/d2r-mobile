# Windows ARM64 原生宿主；与其他平台共享源码表，平台自测不需要游戏文件。
# Native Windows ARM64 host; shares source lists with other platforms; platform tests need no game files.
cmake_minimum_required(VERSION 3.22)
project(dm_windows LANGUAGES C CXX ASM)
if(NOT WIN32 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
  message(FATAL_ERROR "Windows ARM64 is required (use toolchain.cmake with LLVM-MinGW UCRT)")
endif()
if(NOT MINGW OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "LLVM-MinGW Clang is required; MSVC and ARM64EC are not supported")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(DM_BUILD_HOST "Build the experimental complete host (requires generated game headers)" OFF)
option(DM_BUILD_FRONTEND "Compile native window and Vulkan frontend without game files" ON)
option(DM_CHECK_SHARED "Compile shared sources that do not require generated game headers" ON)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(DM "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(HOST "${DM}/host")
set(COMPAT "${CMAKE_CURRENT_SOURCE_DIR}/compat/include")
set(DM_COMPAT_INCLUDES "${COMPAT}" "${DM}/android/compat/include")
set(DM_HOST_FLAGS -fno-strict-aliasing -include "${COMPAT}/dm_windows.h")
add_library(dm_windows_runtime STATIC compat/windows.c compat/mach.c compat/sections.c
  thread_stack.c thread_stack.S audio.c guest_libc.c gate.c)
target_include_directories(dm_windows_runtime PUBLIC ${DM_COMPAT_INCLUDES})
target_compile_definitions(dm_windows_runtime PUBLIC _WIN32_WINNT=0x0A00 _FILE_OFFSET_BITS=64 UNICODE _UNICODE)
target_compile_options(dm_windows_runtime PRIVATE -Wall -Wextra -fno-strict-aliasing)
target_link_libraries(dm_windows_runtime PUBLIC winpthread ktmw32 winmm avrt ole32 uuid psapi bcrypt)
target_link_options(dm_windows_runtime INTERFACE -static)
if(BUILD_TESTING)
  add_executable(dm_windows_audio_test tests/audio.c)
  target_compile_options(dm_windows_audio_test PRIVATE -UNDEBUG)
  target_link_libraries(dm_windows_audio_test PRIVATE dm_windows_runtime)
  add_test(NAME windows-audio COMMAND dm_windows_audio_test)
  add_executable(dm_windows_performance_test tests/performance.c)
  target_compile_options(dm_windows_performance_test PRIVATE -UNDEBUG)
  target_link_libraries(dm_windows_performance_test PRIVATE dm_windows_runtime)
  add_test(NAME windows-performance COMMAND dm_windows_performance_test)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rewrite_blocks.h"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/rewrite.py"
      --emit "${CMAKE_CURRENT_BINARY_DIR}/rewrite_blocks.h"
    DEPENDS tests/rewrite.py tools/asm_coff.py tools/check_arm64.py "${DM}/tools/patch_nso.py" VERBATIM)
  add_executable(dm_windows_runtime_test tests/runtime.c tests/stack_context.cpp "${HOST}/hle_svc.c"
    "${CMAKE_CURRENT_BINARY_DIR}/rewrite_blocks.h")
  target_include_directories(dm_windows_runtime_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${HOST}")
  target_link_libraries(dm_windows_runtime_test PRIVATE dm_windows_runtime)
  target_compile_options(dm_windows_runtime_test PRIVATE ${DM_HOST_FLAGS} -UNDEBUG -ffunction-sections -fdata-sections)
  target_link_options(dm_windows_runtime_test PRIVATE -Wl,--gc-sections)
  add_test(NAME windows-runtime COMMAND dm_windows_runtime_test)
  add_executable(dm_windows_loader_test tests/loader.c "${HOST}/hle.c" "${DM}/android/compat/digest.c")
  target_include_directories(dm_windows_loader_test PRIVATE "${HOST}")
  target_compile_options(dm_windows_loader_test PRIVATE ${DM_HOST_FLAGS} -O1 -UNDEBUG -ffunction-sections -fdata-sections)
  target_link_options(dm_windows_loader_test PRIVATE -Wl,--gc-sections)
  target_link_libraries(dm_windows_loader_test PRIVATE dm_windows_runtime)
  add_test(NAME windows-loader COMMAND dm_windows_loader_test)
  add_test(NAME windows-tools COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/rewrite.py"
    --loader "$<TARGET_FILE:dm_windows_loader_test>")
  add_executable(dm_windows_desktop_test tests/desktop.c)
  target_include_directories(dm_windows_desktop_test PRIVATE "${HOST}")
  target_compile_options(dm_windows_desktop_test PRIVATE ${DM_HOST_FLAGS} -UNDEBUG -ffunction-sections -fdata-sections)
  target_link_options(dm_windows_desktop_test PRIVATE -Wl,--gc-sections)
  target_link_libraries(dm_windows_desktop_test PRIVATE dm_windows_runtime)
  add_test(NAME windows-desktop COMMAND dm_windows_desktop_test --desktop)
  add_test(NAME windows-touch-default COMMAND dm_windows_desktop_test)
  add_executable(dm_windows_mouse_test "${HOST}/tests/t_touch_mac.c" "${HOST}/dm_cursor.c" "${HOST}/dm_hud.c" "${HOST}/dm_prof.c")
  target_include_directories(dm_windows_mouse_test PRIVATE "${HOST}")
  target_compile_options(dm_windows_mouse_test PRIVATE ${DM_HOST_FLAGS} -O1 -UNDEBUG)
  target_link_libraries(dm_windows_mouse_test PRIVATE dm_windows_runtime)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mouse-test/build")
  add_test(NAME windows-mouse COMMAND dm_windows_mouse_test)
  set_tests_properties(windows-mouse PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mouse-test")
endif()
if(DM_BUILD_FRONTEND OR DM_BUILD_HOST)
  include(vulkan.cmake)
  add_subdirectory("${DM}/gfx/vulkan" "${CMAKE_CURRENT_BINARY_DIR}/vulkan")
  target_include_directories(dm_vulkan_engine PRIVATE ${DM_COMPAT_INCLUDES})
  target_compile_options(dm_vulkan_engine PRIVATE "$<$<COMPILE_LANGUAGE:C,CXX>:${DM_HOST_FLAGS}>")
  target_link_libraries(dm_vulkan_engine PUBLIC dm_windows_runtime)
  add_library(dm_windows_frontend STATIC front_windows.c vulkan_present.cpp "${DM}/gfx/vulkan/nvn_vk.c")
  target_include_directories(dm_windows_frontend PRIVATE "${HOST}" "${DM}/gfx/nvn/include")
  target_compile_options(dm_windows_frontend PRIVATE "$<$<COMPILE_LANGUAGE:C,CXX>:${DM_HOST_FLAGS}>")
  target_link_libraries(dm_windows_frontend PUBLIC dm_windows_runtime dm_vulkan_engine user32 gdi32 shell32 xinput)
  if(BUILD_TESTING)
    add_executable(dm_windows_present_test tests/present.cpp)
    target_compile_options(dm_windows_present_test PRIVATE ${DM_HOST_FLAGS} -UNDEBUG)
    target_link_libraries(dm_windows_present_test PRIVATE dm_windows_runtime Vulkan::Vulkan user32)
    add_test(NAME windows-present COMMAND dm_windows_present_test)
    set(ENGINE "${DM}/gfx/vulkan/engine")
    add_executable(dm_windows_native_bc_test "${ENGINE}/tests/native_bc_policy_test.cpp" "${ENGINE}/xbg_blit.cpp")
    add_executable(dm_windows_ldc_fold_test "${ENGINE}/tests/ldc_fold_test.cpp" "${ENGINE}/xbg_spirv_ldc.cpp")
    add_executable(dm_windows_forward_temps_test "${ENGINE}/tests/forward_temps_test.cpp" "${ENGINE}/xbg_spirv_temps.cpp")
    add_executable(dm_windows_pipeline_memo_test "${ENGINE}/tests/pipeline_memo_test.cpp")
    add_executable(dm_windows_writeback_test "${ENGINE}/tests/writeback_skip_test.cpp")
    foreach(test native_bc ldc_fold forward_temps pipeline_memo writeback)
      target_compile_options(dm_windows_${test}_test PRIVATE -UNDEBUG)
      target_include_directories(dm_windows_${test}_test PRIVATE "${DM}/gfx/nvn/include")
      target_link_libraries(dm_windows_${test}_test PRIVATE dm_windows_runtime Vulkan::Vulkan)
      string(REPLACE "_" "-" test_name "${test}")
      add_test(NAME windows-${test_name} COMMAND dm_windows_${test}_test)
    endforeach()
  endif()
endif()
if(NOT DM_BUILD_HOST AND NOT DM_CHECK_SHARED)
  return()
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST}/Makefile" "${DM}/android/tools/host_sources.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${DM}/android/tools/host_sources.py"
  "${HOST}/Makefile" "${CMAKE_CURRENT_BINARY_DIR}/host_sources.cmake" RESULT_VARIABLE parsed)
if(NOT parsed EQUAL 0)
  message(FATAL_ERROR "Cannot parse shared host sources")
endif()
include("${CMAKE_CURRENT_BINARY_DIR}/host_sources.cmake")
list(APPEND HOST_SRCS "${HOST}/touch_mac.c")
file(GLOB HLE_SRCS CONFIGURE_DEPENDS "${HOST}/hle_*.c")
file(GLOB GFX_SRCS CONFIGURE_DEPENDS "${DM}/gfx/*.c" "${DM}/gfx/nvn/*.c" "${DM}/gfx/tex/*.c")
set(ASM_SRCS)
foreach(source "${HOST}/entry.S" "${DM}/gfx/nvn/nvn_entry.S" "${DM}/gfx/nvn/nvn_thunks.S")
  get_filename_component(name "${source}" NAME)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/asm/${name}")
  set(arguments)
  if(name STREQUAL "entry.S")
    list(APPEND arguments --thread-bridge)
  endif()
  add_custom_command(OUTPUT "${output}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/asm_coff.py"
    "${source}" "${output}" ${arguments}
    DEPENDS "${source}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/asm_coff.py" VERBATIM)
  list(APPEND ASM_SRCS "${output}")
endforeach()
if(NOT DM_BUILD_HOST)
  list(FILTER HOST_SRCS EXCLUDE REGEX "/(diag|dm_bright|dm_vault_ui)\\.c$")
  add_library(dm_windows_shared_check OBJECT ${HOST_SRCS} ${HLE_SRCS} ${GFX_SRCS} ${LIBDEFLATE_SRCS} ${ASM_SRCS} main_windows.c)
  target_include_directories(dm_windows_shared_check PRIVATE "${HOST}" "${DM}/gfx/nvn/include")
  target_compile_options(dm_windows_shared_check PRIVATE "$<$<COMPILE_LANGUAGE:C,CXX>:${DM_HOST_FLAGS}>" -O1)
  target_link_libraries(dm_windows_shared_check PRIVATE dm_windows_runtime)
  add_library(dm_windows_main_check OBJECT "${HOST}/main.c")
  target_compile_definitions(dm_windows_main_check PRIVATE main=dm_main)
  target_compile_options(dm_windows_main_check PRIVATE ${DM_HOST_FLAGS})
  target_link_libraries(dm_windows_main_check PRIVATE dm_windows_runtime)
  return()
endif()
foreach(header dm_vault_tab.h dm_hud_exp.h dm_bright_levels.inc)
  if(NOT DM_GEN_DIR OR NOT EXISTS "${DM_GEN_DIR}/${header}")
    message(FATAL_ERROR "Missing ${header}: set DM_GEN_DIR to your own generated headers (see dm/host/Makefile). Use DM_BUILD_HOST=OFF for platform tests.")
  endif()
endforeach()
add_library(dmcore STATIC ${HOST_SRCS} ${HLE_SRCS} ${GFX_SRCS} ${LIBDEFLATE_SRCS} ${ASM_SRCS}
  "${DM}/android/compat/digest.c")
target_include_directories(dmcore PUBLIC "${HOST}" "${DM}/gfx/nvn/include" "${DM_GEN_DIR}")
target_link_libraries(dmcore PUBLIC dm_windows_runtime dm_windows_frontend)
target_compile_options(dmcore PUBLIC "$<$<COMPILE_LANGUAGE:C,CXX>:${DM_HOST_FLAGS}>")
target_compile_options(dmcore PRIVATE -g -Wall -Wno-unused-function)
add_library(dm_host_main OBJECT "${HOST}/main.c")
target_compile_definitions(dm_host_main PRIVATE main=dm_main)
target_link_libraries(dm_host_main PRIVATE dmcore)
add_executable(dmhost main_windows.c $<TARGET_OBJECTS:dm_host_main>)
target_link_libraries(dmhost PRIVATE -Wl,--whole-archive dmcore -Wl,--no-whole-archive)
target_link_options(dmhost PRIVATE -Wl,--stack,16777216 -Wl,--export-all-symbols)
