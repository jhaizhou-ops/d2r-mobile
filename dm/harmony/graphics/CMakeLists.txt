cmake_minimum_required(VERSION 3.18)
project(xbl_harmony_graphics LANGUAGES C CXX)
add_library(xbg_texture STATIC xbg_texture.c xbg_depth.c xbg_framebuffer.c ../compat-r32/gfx/nvn/nvn_tex.c ../compat-r32/gfx/tex/nvn_deswizzle.c)
set_source_files_properties(xbg_texture.c PROPERTIES COMPILE_OPTIONS "-fsigned-char")
target_include_directories(xbg_texture PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(xbg_texture PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(xbg_vulkan STATIC xbg_vulkan.cpp xbg_nvn.cpp xbg_sampler.cpp)
target_include_directories(xbg_vulkan PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(xbg_vulkan PUBLIC cxx_std_17)
target_compile_options(xbg_vulkan PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
set_target_properties(xbg_vulkan PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(xbg_vulkan PUBLIC xbg_texture vulkan)
add_library(xbg_game STATIC xbg_game.cpp xbg_game_nvn.c xbg_shader_key.c xbg_command_clone.c xbg_draw_plan.c xbg_vertex.c xbg_quad.c xbg_stages.cpp xbg_clear.cpp xbg_blit.cpp)
target_include_directories(xbg_game PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ../compat-r32/gfx/nvn/include)
target_compile_features(xbg_game PUBLIC cxx_std_17)
target_compile_options(xbg_game PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Wno-format)
target_compile_options(xbg_game PRIVATE $<$<COMPILE_LANGUAGE:C>:-Werror=incompatible-function-pointer-types>)
# Keep diagnostic/default behavior and assertions; optimize this graphics target only.
# Target flags follow build-type flags so Debug -O0 cannot override this -O2.
target_compile_options(xbg_game PRIVATE -O2)
set_target_properties(xbg_game PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(xbg_game PUBLIC xbg_vulkan)
option(XBG_BUILD_LINK_CHECK "Link an ARM64 shared object to validate the entire library" OFF)
if(XBG_BUILD_LINK_CHECK)
    add_library(xbg_link_check SHARED link_check.cpp)
    target_link_libraries(xbg_link_check PRIVATE -Wl,--whole-archive xbg_game -Wl,--no-whole-archive)
    target_link_options(xbg_link_check PRIVATE -Wl,--no-undefined)
endif()
