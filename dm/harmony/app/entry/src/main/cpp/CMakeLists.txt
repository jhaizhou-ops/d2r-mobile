cmake_minimum_required(VERSION 3.5.0)
project(xblhost LANGUAGES C CXX)
# Optimize every native target (host, graphics, texture decode); keep symbols.
set(CMAKE_C_FLAGS_DEBUG "-O2 -g")
set(CMAKE_CXX_FLAGS_DEBUG "-O2 -g")
set(CMAKE_C_FLAGS_RELEASE "-O2 -g")
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -g")
add_library(xblhost SHARED napi_init.cpp platform_checks.cpp gpu_checks.cpp runtime_bridge.cpp game_graphics.cpp game_keyboard.cpp controller_bridge.cpp game_touch.cpp cloud_sync_bridge.cpp cloud/sw_cloudsync.c)
target_compile_features(xblhost PRIVATE cxx_std_17)
include("${CMAKE_CURRENT_SOURCE_DIR}/real-shader.local.cmake" OPTIONAL)
if(NOT DEFINED XBL_RUNTIME_INCLUDE)
    message(FATAL_ERROR "Prepare an external App project with tools/prepare_app.py first")
endif()
if(DEFINED XBL_RUNTIME_INCLUDE)
    target_include_directories(xblhost PRIVATE "${XBL_RUNTIME_INCLUDE}")
    target_compile_definitions(xblhost PRIVATE XBL_RUNTIME_EXECUTION=1)
    target_sources(xblhost PRIVATE touch_core_session.c touch_core_pulse.c touch_game_mapper.c touch-core/ios/Sources/sw_touch_controls.c touch-core/ios/Sources/sw_touch_game_state.c touch-core/ios/Sources/sw_touch_native_art.c)
endif()
if(NOT DEFINED XBL_HARMONY_SOURCE)
    get_filename_component(XBL_HARMONY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
endif()
add_subdirectory("${XBL_HARMONY_SOURCE}/graphics" "${CMAKE_CURRENT_BINARY_DIR}/xbg")
target_link_libraries(xblhost PRIVATE xbg_vulkan xbg_game)
target_compile_definitions(xblhost PRIVATE XBL_GAME_GRAPHICS=1)
target_compile_options(xblhost PRIVATE -Wall -Wextra)
target_link_libraries(xblhost PUBLIC libace_napi.z.so libdeviceinfo_ndk.z.so libvulkan.so libace_ndk.z.so libnative_window.so libnative_display_manager.so libnative_display_soloist.so)
